// MapLoader.cpp：モデル情報をCSVから読み込む
static void LoadGameModelConfig()
{
    ifstream ifs("model_config.csv");
    if (!ifs)
    {
        printfDx("model_config.csv load failed\n");
        return;
    }

    string line;
    while (getline(ifs, line))
    {
        if (line.empty() || line[0] == '#')
            continue;

        vector<string> cols = SplitCSV(line);
        if (cols.size() < 3 || cols[0] == "id")
            continue;

        int id = stoi(cols[0]);
        string modelPath = cols[2];
        float scaleX = cols.size() >= 5 ? stof(cols[4]) : 1.0f;
        float scaleY = cols.size() >= 6 ? stof(cols[5]) : scaleX;
        float scaleZ = cols.size() >= 7 ? stof(cols[6]) : scaleX;

        // CSV側の情報を元にモデルを登録するため、
        // 新規オブジェクト追加時のコード修正を減らせる
        gameModelHandles[id] = MV1LoadModel(modelPath.c_str());
        if (gameModelHandles[id] != -1)
            MV1SetScale(gameModelHandles[id], VGet(scaleX, scaleY, scaleZ));
    }
}

この作品で伝えたい強み

  • 制作効率を上げるためのツール開発力
    マップをコード上で直接編集する形だと、 変更のたびに確認や修正に時間がかかるため、 配置・保存・読込・当たり判定調整をエディタ上で完結できるようにしました。 プログラマー以外でもマップ調整に関われる構造にし、 チーム制作全体の作業効率を高めることを意識しました。
  • 後から機能追加しやすい設計意識
    FLOOR・WALL・DECO・EVENT・ENEMY・COLLISIONを分けて管理し、 保存形式もセクションごとに分割しました。 敵配置や当たり判定など、後から必要になった機能を追加しても 既存のマップデータや配置処理を壊しにくい構成を意識しています。
  • 調整しやすさを考えた実装
    当たり判定のズレはプレイヤーのストレスに直結するため、 数値だけで管理するのではなく、 BoxCollider風に表示しながら位置やサイズを調整できるようにしました。 実際に遊んで違和感が出た箇所を素早く修正できるため、 デバッグと調整のしやすさを両立しています。
  • プレイヤー体験を意識した裏側の実装
    ミニマップでは探索済みのセルだけを表示し、 未探索エリアを見せすぎないようにすることで、 迷路を探索する緊張感を残す設計にしました。 エディタの使いやすさだけでなく、 最終的にプレイヤーがどう感じるかまで考えて実装しています。
  • 問題を見つけて改善し続ける姿勢
    旧CSVとの互換対応、敵データのEVENTからENEMYへの分離、 未保存時の確認、範囲コピー時の当たり判定保持など、 制作中に出た問題をその場しのぎにせず、 使い続けられる形へ改善してきました。

MapEditor.cpp：マウス座標から配置セルを取得 画面上のマウス位置を3D空間へ変換し、現在の階層で指しているマスを求める処理。 
// マウス位置から現在階層のグリッド座標を計算
VECTOR nearPos = ConvScreenPosToWorldPos(VGet((float)mx, (float)my, 0.0f));
VECTOR farPos  = ConvScreenPosToWorldPos(VGet((float)mx, (float)my, 1.0f));

float targetY = currentLayer * BLOCK_SIZE;
float dy = farPos.y - nearPos.y;

if (fabsf(dy) > 0.0001f)
{
    float t = (targetY - nearPos.y) / dy;

    if (t >= 0.0f && t <= 1.0f)
    {
        VECTOR hit = VGet(0.0f, targetY, 0.0f);
        hit.x = nearPos.x + (farPos.x - nearPos.x) * t;
        hit.z = nearPos.z + (farPos.z - nearPos.z) * t;

        hoverWorldX = hit.x;
        hoverWorldZ = hit.z;
        hoverX = (int)floorf(hit.x / BLOCK_SIZE);
        hoverZ = (int)floorf(hit.z / BLOCK_SIZE);
    }
}

MapIO.cpp：CSV読込処理 CSVのセクション名を見ながら、床・壁・当たり判定・敵などを対応する配列へ復元する処理。 
// MapIO.cpp：CSVを1行ずつ読み込み、現在のセクションに応じて復元
while (getline(ifs, line))
{
    if (line.empty() || line[0] == '#')
        continue;

    if (line[0] == '[')
    {
        section = line;
        continue;
    }

    vector<string> cols = SplitCSV(line);
    if (cols.empty() || cols[0] == "y")
        continue;

    if (section == "[FLOOR]" && cols.size() >= 5)
    {
        int y = stoi(cols[0]);
        int z = stoi(cols[1]);
        int x = stoi(cols[2]);

        if (IsSaveLoadPosValid(y, z, x))
        {
            FloorMap[y][z][x] = stoi(cols[3]);
            FloorRot[y][z][x] = stoi(cols[4]);
        }
    }
    else if (section == "[COLLISION]" && cols.size() >= 4)
    {
        int y = stoi(cols[0]);
        int z = stoi(cols[1]);
        int x = stoi(cols[2]);

        if (IsSaveLoadPosValid(y, z, x))
        {
            CollisionMap[y][z][x] = stoi(cols[3]);
            CollisionBoxOffsetXMap[y][z][x] = cols.size() > 4 ? stoi(cols[4]) : 0;
            CollisionBoxOffsetZMap[y][z][x] = cols.size() > 5 ? stoi(cols[5]) : 0;
            CollisionBoxSizeXMap[y][z][x] = cols.size() > 6 ? stoi(cols[6]) : (int)BLOCK_SIZE;
            CollisionBoxSizeZMap[y][z][x] = cols.size() > 7 ? stoi(cols[7]) : (int)BLOCK_SIZE;
        }
    }
    else if (section == "[ENEMY]" && cols.size() >= 4)
    {
        int y = stoi(cols[0]);
        int z = stoi(cols[1]);
        int x = stoi(cols[2]);

        if (IsSaveLoadPosValid(y, z, x))
            EnemyMap[y][z][x] = NormalizeEnemyIdForCsv(stoi(cols[3]));
    }
}

// MapLoaderMiniMap.inl：プレイヤーがいるセルを訪問済みにする
static void RevealMiniMapAroundPlayer(VECTOR playerPos, int layer)
{
    int playerCellX = WorldToCell(playerPos.x);
    int playerCellZ = WorldToCell(playerPos.z);

    if (IsGameMapPosValid(layer, playerCellZ, playerCellX))
    {
        GameMiniMapVisited[layer][playerCellZ][playerCellX] = true;
    }
}

void DrawMiniMap(VECTOR playerPos)
{
    int layer = GetMapLayerFromWorldY(playerPos.y);
    RevealMiniMapAroundPlayer(playerPos, layer);

    for (int z = 0; z < BLOCK_NUM_Z; z++)
    {
        for (int x = 0; x < BLOCK_NUM_X; x++)
        {
            // 未探索のマスは描画しないため、先の地形が見えない
            if (!GameMiniMapVisited[layer][z][x])
                continue;

            bool walkable = GameFloorMap[layer][z][x] >= 0 ||
                            IsStairsId(GameDecoMap[layer][z][x]);

            if (walkable)
                DrawBox(left + 1, top + 1, right - 1, bottom - 1, GetColor(42, 58, 66), TRUE);

            if (GameCollisionMap[layer][z][x] >= 0)
                DrawBox(left + 2, top + 2, right - 2, bottom - 2, GetColor(120, 55, 70), TRUE);
        }
    }
}

ポケAPI タイピングゲーム

個人制作 / 制作期間：2週間

概要

PokéAPIを利用し、ランダムに取得したポケモン名を ローマ字入力でタイピングするゲームです。

単なる文字列一致ではなく、 状態遷移型ローマ字判定エンジンを独自設計し、 複数入力候補を同時に保持しながら 正誤判定を行う仕組みを実装しました。

技術的特徴

  • 状態遷移型ローマ字判定エンジン（Listによる分岐管理）
  • 「shi / si」「tsu / tu」など複数入力候補の同時追跡
  • 拗音（キャ / シャ）・促音（ッ）対応の2文字先読み処理
  • GASを利用したオンラインランキング送受信機能
  • 仮想キーボード連動によるリアルタイム入力可視化

スクリプト構成・処理内容

  • PokeApiManager.cs
    ランダムIDを生成し、APIからJSONデータを取得。
    取得データを整形し、問題用ポケモン名と画像を生成。
  • TypingEngine.cs
    List<State>を用いた状態遷移型入力判定を実装。
    入力ごとに候補を絞り込み、 分岐入力を許容しながら正誤判定を行う。
  • RankingManager.cs
    GASへPOST送信しスコアを保存。
    mode=getでランキング取得後、 スコア順に並び替えて表示。
  • UIController.cs
    入力状態に応じて文字色を変更し、 次に押すべきキーをガイド表示。

工夫した点

  • 単純一致ではなく「入力状態管理型エンジン」として設計
  • 複数候補を同時保持することで柔軟な入力対応を実現
  • 非同期通信時のエラー処理を実装し安定性を確保
  • 入力ミス時の演出を追加し、ゲーム体験を向上
